cmake_minimum_required(VERSION 3.22)
project(brlinst VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
endif()
include(CTest)
# Support both misc/tools/brlinst and a standalone checkout beside brlcad.
get_filename_component(_brlinst_brlcad_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(NOT EXISTS "${_brlinst_brlcad_source}/src/libbu/json.hpp")
  get_filename_component(_brlinst_brlcad_source "${CMAKE_CURRENT_SOURCE_DIR}/../brlcad" ABSOLUTE)
endif()
get_filename_component(_brlinst_bext_default "${_brlinst_brlcad_source}/../bext" ABSOLUTE)
set(BRLINST_BEXT_ROOT "${_brlinst_bext_default}" CACHE PATH "BRL-CAD external dependency checkout")
set(BRLINST_ZLIB_SOURCE "${BRLINST_BEXT_ROOT}/zlib/zlib" CACHE PATH "Unmodified zlib sources")
set(BRLINST_JSON_INCLUDE "${_brlinst_brlcad_source}/src/libbu" CACHE PATH "Directory containing nlohmann json.hpp")
if(NOT EXISTS "${BRLINST_JSON_INCLUDE}/json.hpp")
  message(FATAL_ERROR "Set BRLINST_JSON_INCLUDE to the directory containing nlohmann json.hpp (3.11 or later)")
endif()
# Build bext sources without altering them or inheriting its symbol-prefix setup.
if(EXISTS "${BRLINST_ZLIB_SOURCE}/zlib.h")
  set(zlib_sources adler32.c compress.c crc32.c deflate.c inflate.c inffast.c inftrees.c trees.c uncompr.c zutil.c)
  list(TRANSFORM zlib_sources PREPEND "${BRLINST_ZLIB_SOURCE}/")
  add_library(brlinst_zlib STATIC ${zlib_sources})
  target_include_directories(brlinst_zlib PUBLIC "${BRLINST_ZLIB_SOURCE}")
else()
  set(ZLIB_USE_STATIC_LIBS ON)
  find_package(ZLIB REQUIRED)
  add_library(brlinst_zlib ALIAS ZLIB::ZLIB)
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/COPYING" BRLINST_NOTICES)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY.txt" _brlinst_dependency_notices)
string(APPEND BRLINST_NOTICES "\n\n${_brlinst_dependency_notices}")
set(BRLINST_EXTRA_NOTICES "" CACHE FILEPATH "Additional notices for the selected compiler/runtime libraries")
if(BRLINST_EXTRA_NOTICES)
  file(READ "${BRLINST_EXTRA_NOTICES}" _brlinst_extra_notices)
  string(APPEND BRLINST_NOTICES "\n\n${_brlinst_extra_notices}")
endif()
# Byte arrays avoid MSVC string-literal limits and preserve license text exactly.
string(HEX "${BRLINST_NOTICES}" _brlinst_notice_hex)
string(REGEX REPLACE "(................................................................)" "\\1\n" _brlinst_notice_hex "${_brlinst_notice_hex}")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," BRLINST_NOTICES_BYTES "${_brlinst_notice_hex}")
configure_file(resources/notices.hpp.in generated/notices.hpp @ONLY)
add_library(brlinst_core STATIC src/core.cpp src/input.cpp src/package.cpp src/transaction.cpp)
target_include_directories(brlinst_core PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(brlinst_core SYSTEM PUBLIC "${BRLINST_JSON_INCLUDE}")
target_link_libraries(brlinst_core PUBLIC brlinst_zlib)
if(MSVC)
  target_compile_options(brlinst_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(brlinst_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
# The portable model allows native tests of authoring rules. MSI serialization
# and custom actions are built only for Windows; they use the operating system.
option(BRLINST_WITH_MSI "Enable Windows-native MSI generation" ${WIN32})
if(BRLINST_WITH_MSI AND NOT WIN32)
  message(FATAL_ERROR "BRLINST_WITH_MSI requires a Windows target")
endif()
if(BRLINST_WITH_MSI OR BUILD_TESTING)
  add_library(brlinst_msi_model STATIC src/msi_model.cpp)
  target_link_libraries(brlinst_msi_model PUBLIC brlinst_core)
endif()
add_executable(brlinst src/pack.cpp)
target_link_libraries(brlinst PRIVATE brlinst_core)
if(WIN32)
  enable_language(RC)
  target_compile_definitions(brlinst_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0a00)
  target_sources(brlinst_core PRIVATE src/resources.cpp)
  target_link_libraries(brlinst PRIVATE version)
  if(MINGW)
    target_link_options(brlinst PRIVATE -municode -static)
  endif()
  if(BRLINST_WITH_MSI)
    target_sources(brlinst PRIVATE src/msi.cpp)
    target_compile_definitions(brlinst PRIVATE BRLINST_WITH_MSI=1)
    target_link_libraries(brlinst PRIVATE brlinst_msi_model msi cabinet bcrypt ole32)
    add_library(brlinst-msi-actions SHARED src/msi_actions.cpp)
    set_target_properties(brlinst-msi-actions PROPERTIES PREFIX "")
    target_link_libraries(brlinst-msi-actions PRIVATE brlinst_core msi shell32 ole32 uuid)
    if(MINGW)
      target_link_options(brlinst-msi-actions PRIVATE -static)
    endif()
    if(MSVC)
      target_compile_options(brlinst-msi-actions PRIVATE /W4 /utf-8)
    endif()
    add_dependencies(brlinst brlinst-msi-actions)
    install(TARGETS brlinst-msi-actions RUNTIME DESTINATION bin)
  endif()
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "brlinst requires an x64 toolchain")
  endif()
  foreach(runtime IN ITEMS brlinst-stub brlinst-uninstall)
    add_executable(${runtime} WIN32 src/windows.cpp src/wizard.cpp src/runtime.cpp resources/runtime.rc)
    target_link_libraries(${runtime} PRIVATE brlinst_core comctl32 shell32 ole32 uuid advapi32 user32 gdi32)
    if(MINGW)
      target_link_options(${runtime} PRIVATE -municode -static)
    endif()
    if(MSVC)
      target_compile_options(${runtime} PRIVATE /W4 /utf-8)
      target_link_options(${runtime} PRIVATE /MANIFEST:NO)
    endif()
  endforeach()
  target_compile_definitions(brlinst-uninstall PRIVATE BRLINST_UNINSTALL=1)
  add_dependencies(brlinst brlinst-stub brlinst-uninstall)
  install(TARGETS brlinst brlinst-stub brlinst-uninstall RUNTIME DESTINATION bin)
else()
  install(TARGETS brlinst RUNTIME DESTINATION bin)
endif()
install(FILES cmake/Brlinst.cmake cmake/Package.cmake.in cmake/BRLCAD.cmake DESTINATION share/brlinst/cmake)
install(FILES COPYING THIRD_PARTY.txt README.adoc DESTINATION share/brlinst)
install(FILES examples/brlcad.json.in DESTINATION share/brlinst/examples)
if(BUILD_TESTING)
  add_executable(brlinst-msi-model-tests tests/msi_model_tests.cpp)
  target_link_libraries(brlinst-msi-model-tests PRIVATE brlinst_msi_model)
  if(MINGW)
    target_link_options(brlinst-msi-model-tests PRIVATE -static)
  endif()
  add_test(NAME brlinst-msi-model COMMAND brlinst-msi-model-tests)
  add_executable(brlinst-tests tests/core_tests.cpp)
  target_link_libraries(brlinst-tests PRIVATE brlinst_core)
  if(WIN32)
    add_dependencies(brlinst-tests brlinst-stub brlinst-uninstall)
    if(MINGW)
      target_link_options(brlinst-tests PRIVATE -static)
    endif()
  endif()
  add_test(NAME brlinst-core COMMAND brlinst-tests)
  if(WIN32 AND BRLINST_WITH_MSI)
    find_program(BRLINST_POWERSHELL NAMES powershell pwsh)
    if(BRLINST_POWERSHELL)
      add_test(NAME brlinst-msi-package COMMAND "${BRLINST_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/windows-msi-smoke.ps1"
        -BuildDirectory "$<TARGET_FILE_DIR:brlinst>")
    endif()
  endif()
  add_test(NAME brlinst-cpack COMMAND "${CMAKE_COMMAND}"
    "-DBRLINST=$<TARGET_FILE:brlinst>"
    "-DFIXTURE_TOOL=$<TARGET_FILE:brlinst-tests>"
    "-DMODULE_DIR=${CMAKE_CURRENT_SOURCE_DIR}/cmake"
    "-DTEST_ROOT=${CMAKE_CURRENT_BINARY_DIR}/cpack-test"
    "-DTEST_GENERATOR=${CMAKE_GENERATOR}"
    "-DTEST_MSI=${BRLINST_WITH_MSI}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cpack_test.cmake")
endif()
